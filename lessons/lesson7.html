  • 속성: 객체의 특성 (변수)

def __init__(self, name, model="Standard"):
    self.name = name
    self.model = model
    self._battery = 100
    self.position = {"x": 0, "y": 0}
    self.status = "대기"
    MechDog.total_dogs += 1

def __str__(self):
    return f"🐕 {self.name}({self.model}) - 배터리:{self._battery}%"

def bark(self):
    if self._battery > 5:
        print(f"{self.name}: 멍멍! 🐕")
        self._battery -= 5
        self.status = "짖음"
    else:
        print(f"{self.name}: 배터리 부족... 😴")
