🔧 실제 MechDog 색상 인식 코드

# main.py - 실제 MechDog 색상 인식 코드
import Hiwonder
import time
import Hiwonder_IIC
from HW_MechDog import MechDog

# 객체 초기화
iic2 = Hiwonder_IIC.IIC(2)
cam = Hiwonder_IIC.ESP32S3Cam(iic2)  # 카메라
i2c1 = Hiwonder_IIC.IIC(1)
i2csonar = Hiwonder_IIC.I2CSonar(i2c1)  # LED 제어용
mechdog = MechDog()

time.sleep(5)

while True:
    color_list = cam.color_recognition()  # 색상 인식
    print(color_list)
    
    if cam.RED in color_list:  # 빨간색 감지
        i2csonar.setRGB(0, 250, 0, 0)  # 빨간색 LED
    
    if cam.GREEN in color_list:  # 초록색 감지
        i2csonar.setRGB(0, 0, 250, 0)  # 초록색 LED
    
    if cam.BLUE in color_list:  # 파란색 감지
        i2csonar.setRGB(0, 0, 0, 250)  # 파란색 LED
    
    time.sleep(0.1)
